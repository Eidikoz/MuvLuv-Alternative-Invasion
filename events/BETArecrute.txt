country_event = {
	id = BETArecrute.1
	title = "BETArecrute.1.name"
	desc = "BETArecrute.1.desc"
	hidden = yes


	trigger = {
		AND ={ 
			has_global_flag = BET_sinkiang
			has_army_manpower = {
				size < 12000000
			}
			BET = { exists = yes }
			tag = BET
		}
	}
	mean_time_to_happen = { days = 5 }

	option = {
		name = BETAInvasion
		BET = {
			load_oob = "BET_sinkiang"
		}
	}
}
